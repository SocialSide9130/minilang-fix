program = expr ;
expr = assign ;
assign = identifier "=" add ;
add = mul { "+" | "-" mul } ;
mul = unary { "*" | "/" | "%" unary } ;
unary = [ "+" | "-" ] primary ;
primary = number | "(" expr ")" | variable | functioncall ;
number = { digit } ;
identifier = alphabet_, { alphabet_ | digit } ;
alphabet = "A".."Z" | "a".."z" ;
alphabet_ = alphabet | "_" ; 
digit = "0".."9" ;
