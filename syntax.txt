program = expr ;
expr = assign ;
assign = identifier "=" add ;
add = mul { "+" | "-" mul } ;
mul = elem { "*" | "/" | "%" elem } ;
elem = number | "(" expr ")" | variable | functioncall ;
number = [ "-" ], digit, { digit } ;
identifier = alphabet_, { alphabet_ | digit } ;
alphabet = "A".."Z" | "a".."z" ;
alphabet_ = alphabet | "_" ; 
digit = "0".."9" ;
